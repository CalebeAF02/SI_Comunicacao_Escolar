\section{Delimitação inicial}
\subsection{Dentro do escopo}
\begin{itemize}
\item Comunicação institucional entre direção, coordenação, secretária, professores, alunos e responsáveis.
\item Organização, moderação, distribuição e consulta de avisos escolares.
\item Categorização e segmentação por público e canal.
\item Registro de avaliação, recebimento e visualização das comunicações.
\end{itemize}

\subsection{Fora do escopo}
\begin{itemize}
\item Gestão financeira da instituição.
\item Gestão de funcionários, matrícula e recursos administrativos.
\item Substituição integral dos processos pedagógicos ou administrativos.
\end{itemize}

\section{Questões em aberto}
\begin{enumerate}
\item Como definir os dados mínimos necessários para cada tipo de comunicação?
\item Quais categorias exigem obrigatoriamente moderação?
\item Como configurar turmas, responsáveis, canais e critérios de segmentação?
\item Qual nível de confirmação de leitura é necessário para avisos críticos?
\item Que funcionalidades devem ser priorizadas nas próximas entregas?
\end{enumerate}
